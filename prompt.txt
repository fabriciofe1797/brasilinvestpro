                totalInvestido += aporteReais;

                if (reinvestir) {
                    const dividendos = patrimonio * taxaMensal;
                    patrimonio += dividendos;
                }
                
                // Registra dados anuais
                if (i % 12 === 0) {
                    dadosAnuais.push({
                        ano: anoAtual + (i / 12),
                        valor: patrimonio / cambioAtual // Converte de volta para Euro
                    });
                }
            }
            
            const patrimonioFinalEuro = patrimonio / cambioAtual;
            const dividendosMensaisEuro = (patrimonio * taxaMensal) / cambioAtual;
            const retorno = ((patrimonioFinalEuro - (totalInvestido / cambioAtual)) / (totalInvestido / cambioAtual)) * 100;
            
            // Atualiza UI
            document.getElementById('resultados').style.display = 'block';
            document.getElementById('patrimonioFinal').textContent = '€ ' + patrimonioFinalEuro.toLocaleString('pt-BR', {maximumFractionDigits: 0});
            document.getElementById('patrimonioFinalRS').textContent = 'R$ ' + patrimonio.toLocaleString('pt-BR', {maximumFractionDigits: 0});
            document.getElementById('totalInvestido').textContent = '€ ' + (totalInvestido / cambioAtual).toLocaleString('pt-BR', {maximumFractionDigits: 0});
            document.getElementById('dividendosMes').textContent = '€ ' + dividendosMensaisEuro.toLocaleString('pt-BR', {maximumFractionDigits: 0});
            document.getElementById('dividendosMesRS').textContent = 'R$ ' + (patrimonio * taxaMensal).toLocaleString('pt-BR', {maximumFractionDigits: 0});
            document.getElementById('retornoTotal').textContent = '+' + retorno.toFixed(1) + '%';
            
            // Atualiza gráfico
            atualizarGrafico(dadosAnuais);
            
            // Atualiza progress ring (meta: €2000/mês)
            const meta = 2000;
            const percentual = Math.min((dividendosMensaisEuro / meta) * 100, 100);
            const circunferencia = 2 * Math.PI * 52;
            const offset = circunferencia - (percentual / 100) * circunferencia;
            
            document.getElementById('progressCircle').style.strokeDashoffset = offset;
            document.getElementById('progressPercent').textContent = percentual.toFixed(1) + '%';
            
            // Scroll para resultados
            document.getElementById('resultados').scrollIntoView({ behavior: 'smooth' });
        }

        function calculateProjection() {
